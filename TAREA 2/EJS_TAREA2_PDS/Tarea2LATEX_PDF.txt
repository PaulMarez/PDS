import numpy as np
import matplotlib.pyplot as plt

def correlacion_cruzada(x, y):
    Nx, Ny = len(x), len(y)
    R = []
    lags = range(-(Ny-1), Nx)
    for k in lags:
        suma = 0
        for n in range(Nx):
            if 0 <= n+k < Ny:
                suma += x[n] * y[n+k]
        R.append(suma)
    return np.array(R), np.array(list(lags))

def detect_frequencies(x, fs, frequencies):
    N = len(x); n = np.arange(N); medidas = []
    for f in frequencies:
        r_sin = np.sin(2*np.pi*f*n/fs)
        r_cos = np.cos(2*np.pi*f*n/fs)
        Rs, _ = correlacion_cruzada(x, r_sin)
        Rc, _ = correlacion_cruzada(x, r_cos)
        mag = np.sqrt(np.max(np.abs(Rs))**2 + np.max(np.abs(Rc))**2)
        medidas.append(mag)
    return np.array(medidas)

fs = 100; t = np.arange(0, 2, 1/fs)
x = (1.5*np.sin(2*np.pi*4*t) + 0.7*np.sin(2*np.pi*9*t + np.pi/4)
     + 2.0*np.sin(2*np.pi*16*t - np.pi/3))
frecs = list(range(1,21)); medidas = detect_frequencies(x, fs, frecs)

plt.figure(); plt.plot(t, x)
plt.title("Senal de prueba"); plt.xlabel("Tiempo (s)"); plt.ylabel("Amplitud"); plt.grid(True)
plt.savefig("ej10_x.png", dpi=150, bbox_inches='tight'); plt.close()

plt.figure(); plt.stem(frecs, medidas, basefmt=" ")
